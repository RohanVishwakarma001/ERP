import asyncio, botocore, random

from aiobotocore.session import get_session


class ResilientS3Uploader:

    def __init__(self, bucket_name: str, max_concurrency: int = 16):

        self.bucket = bucket_name

        self.sem = asyncio.Semaphore(max_concurrency)

        self.session = get_session()


    async def upload_with_backoff(self, key: str, payload: bytes, max_retries: int = 5):

        async with self.sem:

            for attempt in range(max_retries):

                try:

                    async with self.session.create_client('s3') as client:

                        await client.put_object(Bucket=self.bucket, Key=key, Body=payload)

                        return True

                except botocore.exceptions.ClientError as e:

                    jitter = random.uniform(0, 0.5)

                    await asyncio.sleep((2 ** attempt) + jitter)

            await self.push_to_dlq(key, payload)
